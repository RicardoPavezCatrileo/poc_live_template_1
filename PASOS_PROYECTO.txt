# DriverFactory: crea/cierra el navegador (ThreadLocal para paralelo)
# Soporta: chrome (default), firefox, edge

# Contenido de DriverFactory.java:
package com.qa.core;

import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.chrome.ChromeOptions;
import org.openqa.selenium.edge.EdgeDriver;
import org.openqa.selenium.edge.EdgeOptions;
import org.openqa.selenium.firefox.FirefoxDriver;
import org.openqa.selenium.firefox.FirefoxOptions;

public final class DriverFactory {
    private static final ThreadLocal<WebDriver> DRIVER = new ThreadLocal<>();

    private DriverFactory() {}

    public static WebDriver getDriver() {
        if (DRIVER.get() == null) {
            DRIVER.set(crearDriver());
        }
        return DRIVER.get();
    }

    private static WebDriver crearDriver() {
        WebDriver driver = switch (Config.BROWSER.toLowerCase()) {
            case "firefox" -> {
                FirefoxOptions o = new FirefoxOptions();
                if (Config.HEADLESS) o.addArguments("-headless");
                yield new FirefoxDriver(o);
            }
            case "edge" -> {
                EdgeOptions o = new EdgeOptions();
                if (Config.HEADLESS) o.addArguments("--headless=new");
                yield new EdgeDriver(o);
            }
            default -> {
                ChromeOptions o = new ChromeOptions();
                if (Config.HEADLESS) o.addArguments("--headless=new");
                yield new ChromeDriver(o);
            }
        };
        driver.manage().window().maximize();
        return driver;
    }

    public static void quitDriver() {
        if (DRIVER.get() != null) {
            DRIVER.get().quit();
            DRIVER.remove();
        }
    }
}

package com.qa.pages;

import com.qa.core.Config;
import org.openqa.selenium.By;
import org.openqa.selenium.JavascriptExecutor;
import org.openqa.selenium.TimeoutException;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

import java.time.Duration;

public abstract class BasePage {
    protected final WebDriver driver;
    protected final WebDriverWait wait;

    protected BasePage(WebDriver driver) {
        this.driver = driver;
        this.wait = new WebDriverWait(driver, Duration.ofSeconds(Config.TIMEOUT));
    }

    /** Ruta relativa de la pagina, ej: "/login". Vacio = home. */
    protected abstract String ruta();

    public void abrir() {
        driver.get(Config.BASE_URL + ruta());
        esperarCarga();
    }

    protected void esperarCarga() {
        wait.until(d -> "complete".equals(
                ((JavascriptExecutor) d).executeScript("return document.readyState")));
    }

    protected WebElement esperarVisible(By locator) {
        return wait.until(ExpectedConditions.visibilityOfElementLocated(locator));
    }

    protected void click(By locator) {
        wait.until(ExpectedConditions.elementToBeClickable(locator)).click();
    }

    protected void escribir(By locator, String texto) {
        WebElement e = esperarVisible(locator);
        e.clear();
        e.sendKeys(texto);
    }

    protected String obtenerTexto(By locator) {
        return esperarVisible(locator).getText();
    }

    protected boolean estaVisible(By locator) {
        try {
            return esperarVisible(locator).isDisplayed();
        } catch (TimeoutException e) {
            return false;
        }
    }

    public String getTitulo() {
        return driver.getTitle();
    }

    public String getUrlActual() {
        return driver.getCurrentUrl();
    }
}

package com.qa.pages;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;

public class LoginPage extends BasePage {

    // Locators
    private final By txtUsuario = By.id("username");
    private final By txtClave = By.id("password");
    private final By btnLogin = By.cssSelector("button[type='submit']");
    private final By lblMensaje = By.id("flash");

    public LoginPage(WebDriver driver) {
        super(driver);
    }

    @Override
    protected String ruta() {
        return "/login";
    }

    // Acciones
    public void login(String usuario, String clave) {
        escribir(txtUsuario, usuario);
        escribir(txtClave, clave);
        click(btnLogin);
    }

    public String obtenerMensaje() {
        return obtenerTexto(lblMensaje);
    }
}

package com.qa.tests;

import com.qa.core.DriverFactory;
import org.openqa.selenium.WebDriver;
import org.testng.annotations.AfterMethod;
import org.testng.annotations.BeforeMethod;

public abstract class BaseTest {
    protected WebDriver driver;

    @BeforeMethod(alwaysRun = true)
    public void setUp() {
        driver = DriverFactory.getDriver();
    }

    @AfterMethod(alwaysRun = true)
    public void tearDown() {
        DriverFactory.quitDriver();
    }
}

package com.qa.tests;

import com.qa.pages.LoginPage;
import org.testng.Assert;
import org.testng.annotations.Test;

public class LoginTest extends BaseTest {

    @Test(description = "Login exitoso con credenciales validas")
    public void loginExitoso() {
        LoginPage login = new LoginPage(driver);
        login.abrir();
        login.login("tomsmith", "SuperSecretPassword!");

        Assert.assertTrue(login.obtenerMensaje().contains("You logged into a secure area!"),
                "No aparecio el mensaje de login exitoso");
        Assert.assertTrue(login.getUrlActual().contains("/secure"), "No redirigio a /secure");
    }

    @Test(description = "Login fallido con usuario invalido")
    public void loginFallido() {
        LoginPage login = new LoginPage(driver);
        login.abrir();
        login.login("usuarioMalo", "claveMala");

        Assert.assertTrue(login.obtenerMensaje().contains("Your username is invalid!"),
                "No aparecio el mensaje de error");
    }
}

PASO 9: Flujo de trabajo con ramas y Pull Request (PR)
----------------------------------------------------------
# PR (GitHub / Azure DevOps) = MR, Merge Request (GitLab). Es lo mismo:
# solicitud para unir los cambios de una rama a otra (normalmente main).
#
# CICLO: rama -> cambios -> push -> PR -> revision -> merge -> pull

# 9.1 Crear rama nueva y moverse a ella (-b = crear + moverse)
git checkout -b feature/nombre-del-cambio

# Ver en que rama estoy (la del * es la actual)
git branch

# Moverse entre ramas existentes (sin -b)
git checkout main
git checkout feature/nombre-del-cambio
# (o git switch <rama>; o clic en la rama en la barra azul de VS Code)

# OJO: si la barra de VS Code muestra otra rama que la terminal,
# manda la TERMINAL. Recargar: Ctrl+Shift+P -> Developer: Reload Window

# 9.2 Hacer cambios (ej: crear README.md)
code README.md

# 9.3 Commit y subir la rama
git add README.md
git commit -m "Agrega README"
git push -u origin feature/nombre-del-cambio

# 9.4 Crear el PR en GitHub
#   - Boton amarillo "Compare & pull request"
#     (o pestana Pull requests -> New pull request)
#   - Verificar: base: main  <-  compare: feature/...
#   - Titulo + descripcion -> Create pull request

# 9.5 Revisar y hacer merge
#   - Pestana "Files changed": ver exactamente que cambio
#   - Conversation -> Merge pull request -> Confirm merge
#   - Delete branch (borra la rama SOLO en GitHub)

# 9.6 Actualizar main local y limpiar
git checkout main
git pull
git branch -d feature/nombre-del-cambio
git fetch --prune
git branch
-> Debe mostrar solo: * main

# Nota: si GitHub aparece en espanol ("principal", "Solicitudes de
# extraccion") es Google Traductor de Chrome -> "Mostrar original".

COMO REUTILIZAR ESTE TEMPLATE EN UN PROYECTO NUEVO
----------------------------------------------------------
1. git clone https://github.com/RicardoPavezCatrileo/poc_live_template_1.git
   (o copiar la carpeta completa sin target/ ni .git/)
2. En pom.xml cambiar <artifactId> por el nombre del proyecto.
3. En Config.java cambiar BASE_URL (o usar -DbaseUrl=...).
4. Borrar LoginPage / LoginTest de ejemplo si no sirven.
5. Abrir en VS Code (File -> Open Folder) y ejecutar: mvn test
6. Si VS Code marca rojo pero mvn compila: Ctrl+Shift+P ->
   "Java: Clean Java Language Server Workspace" -> Reload and delete
